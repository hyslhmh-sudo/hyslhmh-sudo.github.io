function ishAbierto_() { return PropertiesService.getScriptProperties().getProperty('ISH_MODO') === 'abierto'; }
function menuModoIsh() {
  const ui = SpreadsheetApp.getUi(), p = PropertiesService.getScriptProperties();
  const abierto = ishAbierto_();
  const r = ui.alert('ISH', abierto
    ? 'Hoy el ISH está ABIERTO para las personas con el módulo ISH en ACCESOS. ¿Cerrarlo? (Administración sigue entrando.)'
    : 'Hoy el ISH es SOLO para Administración. ¿Abrirlo a las personas que tengan «ISH» en la columna Módulos de ACCESOS?', ui.ButtonSet.YES_NO);
  if (r !== ui.Button.YES) return;
  p.setProperty('ISH_MODO', abierto ? 'cerrado' : 'abierto');
  ui.alert('Listo: el ISH quedó ' + (abierto ? 'solo para Administración.' : 'abierto para quienes tengan el módulo ISH.'));
}

function ishSubNombre_(c) { const s = ISH_SUB.filter(x => x[0] === c)[0]; return s ? c + ' · ' + s[2] : c; }
function ishModNombre_(m) { const s = ISH.MODULOS.filter(x => x[0] === m)[0]; return s ? m + ' · ' + s[1] : m; }

/* =========================================================
 *  INSTALACIÓN: arma la planilla ISH de cada red listada en REDES ISH
 * ========================================================= */
function ishInstalar() {
  const libro = SpreadsheetApp.getActive();
  let rp = libro.getSheetByName(ISH.HOJA_REDES);
  if (!rp) {
    rp = libro.insertSheet(ISH.HOJA_REDES);
    rp.getRange(1, 1, 1, 3).setValues([['Red', 'Planilla ISH (enlace o ID)', 'Estado']]).setFontWeight('bold');
    avisar_('Creé la hoja ' + ISH.HOJA_REDES + '. Completá una fila por red con el enlace de su planilla ISH y Estado «Activa», y volvé a correr el menú.');
    return;
  }
  const starh = {};
  redes_().forEach(r => starh[r.nombre] = r.id);
  const filas = rp.getDataRange().getDisplayValues().slice(1).filter(x => x[0] && x[1]);
  const informe = [];
  filas.forEach(x => {
    const red = x[0].trim(), id = idPlanilla_(x[1]);
    const ss = SpreadsheetApp.openById(id);
    const rel = ss.getSheetByName('RELEVAMIENTO');
    if (rel && rel.getLastRow() > 1 && rel.getRange(2, 9, rel.getLastRow() - 1, 1).getValues().some(v => v[0] !== '')) {
      informe.push(red + ': ya tiene datos cargados, no se tocó.'); return;
    }
    if (!starh[red]) throw new Error('No encuentro la planilla STAR-H de ' + red + ' en REDES (de ahí salen los hospitales).');
    const est = SpreadsheetApp.openById(starh[red]).getSheetByName('ESTABLECIMIENTOS').getDataRange().getDisplayValues().slice(1)
      .filter(e => e[0] && /^hospital$/i.test(String(e[2]).trim()));
    if (!est.length) { informe.push(red + ': no tiene establecimientos con Tipo «Hospital».'); return; }
    ishArmar_(ss, red, est);
    informe.push(red + ': ' + est.length + ' hospital(es) · ' + (est.length * ISH_ITEMS.length) + ' ítems · ' + (est.length * ISH_AMEN.length) + ' amenazas.');
  });
  avisar_(informe.join('\n') || 'No hay redes en ' + ISH.HOJA_REDES + '.');
}

function ishRedes_() {
  const sh = SpreadsheetApp.getActive().getSheetByName(ISH.HOJA_REDES);
  if (!sh) return [];
  return sh.getDataRange().getDisplayValues().slice(1)
    .filter(x => x[0] && x[1] && /^activa$/i.test(String(x[2]).trim()))
    .map(x => ({ nombre: x[0].trim(), id: idPlanilla_(x[1]) }));
}
function ishRelObj_(x) {
  return { codigo: x[0], estab: x[1], red: x[2], tipo: x[3], n: Number(x[4]), modulo: String(x[5]).charAt(0), sub: String(x[6]).split(' · ')[0], item: x[7],
           nivel: x[8], obs: x[12], fecha: x[13], evaluador: x[14] };
}
function ishAmObj_(x) {
  return { codigo: x[0], id: x[2], cod: x[3], grupo: x[4], amenaza: x[5], nivel: x[6], prepararse: x[7], obs: x[8], fecha: x[9], evaluador: x[10] };
}
function ishPesos_(ss) {
  const pe = ss.getSheetByName('PESOS');
  const num = v => { const n = Number(String(v).replace(',', '.')); return isNaN(n) ? 0 : n; };
  const val = pe.getRange(3, 1, 3, 2).getValues();
  const mod = pe.getRange(8, 1, 2, 4).getValues();
  const sub = pe.getRange(12, 1, ISH_SUB.length, 4).getValues();
  const o = { valores: {}, modelos: {}, sub: {} };
  val.forEach(v => o.valores[v[0]] = num(v[1]));
  mod.forEach(v => o.modelos[v[0]] = { '2': num(v[1]), '3': num(v[2]), '4': num(v[3]) });
  sub.forEach(v => o.sub[String(v[0])] = num(v[3]));
  return o;
}
function ishLeer_(u, r, conTodo) {
  const ss = SpreadsheetApp.openById(r.id);
  const est = ss.getSheetByName('ESTABLECIMIENTOS').getDataRange().getDisplayValues().slice(1)
    .filter(x => x[0] && veEstab_(u, x[0]))
    .map(x => ({ codigo: x[0], tipo: x[2], nombre: x[3], coordinador: x[4], fecha: x[5], estado: x[6], modelo: x[7] || 'Modelo 1', puedeCargar: cargaEstab_(u, r.nombre, x[0]) }));
  if (!est.length) return null;
  const cods = est.map(x => x.codigo);
  const pesoItem = {};
  ss.getSheetByName('ITEMS').getDataRange().getValues().slice(1).forEach(x => { if (x[0] !== '') pesoItem[x[0]] = Number(x[4]) || 0; });
  const rel = ss.getSheetByName('RELEVAMIENTO').getDataRange().getDisplayValues().slice(1).filter(x => x[0] && cods.indexOf(x[0]) >= 0).map(ishRelObj_);
  const am = ss.getSheetByName('AMENAZAS').getDataRange().getDisplayValues().slice(1).filter(x => x[0] && cods.indexOf(x[0]) >= 0).map(ishAmObj_);
  return { nombre: r.nombre, establecimientos: est, pesos: ishPesos_(ss), pesoItem: pesoItem, relevamiento: rel, amenazas: am };
}
function ishCatalogo_(u) {
  exigirModulo_(u, 'ISH');
  const salida = { ok: true, usuario: publico_(u), generado: new Date().toISOString(), modulos: ISH.MODULOS, submodulos: ISH_SUB, redes: [] };
  ishRedes_().filter(r => veRed_(u, r.nombre)).forEach(r => { const x = ishLeer_(u, r); if (x) salida.redes.push(x); });
  return salida;
}
function ishTablero_(u) { return ishCatalogo_(u); }

function ishGuardar_(u, items) {
  exigirModulo_(u, 'ISH');
  if (!items.length) return { ok: true, resultados: [] };
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  const resultados = [];
  const hoy = Utilities.formatDate(new Date(), 'America/Argentina/Buenos_Aires', 'dd/MM/yyyy');
  try {
    const mapa = {}; ishRedes_().forEach(r => mapa[r.nombre] = r);
    const porRed = {}; items.forEach(it => (porRed[it.red] = porRed[it.red] || []).push(it));
    Object.keys(porRed).forEach(red => {
      const r = mapa[red];
      if (!r) { porRed[red].forEach(it => resultados.push({ uid: it.uid, ok: false, error: 'Red no activa: ' + red })); return; }
      const ss = SpreadsheetApp.openById(r.id);
      const rl = ss.getSheetByName('RELEVAMIENTO'), am = ss.getSheetByName('AMENAZAS');
      const fR = {}; rl.getRange(2, 1, Math.max(rl.getLastRow() - 1, 1), 5).getDisplayValues().forEach((k, i) => { if (k[0]) fR[k[0] + '|' + k[4]] = i + 2; });
      const fA = {}; am.getRange(2, 1, Math.max(am.getLastRow() - 1, 1), 3).getDisplayValues().forEach((k, i) => { if (k[0]) fA[k[0] + '|' + k[2]] = i + 2; });
      const tocadas = [];
      porRed[red].forEach(it => {
        try {
          if (!cargaEstab_(u, red, it.codigo)) throw new Error('Sin permiso de carga en ' + it.codigo);
          const fecha = String(it.fecha || hoy).slice(0, 20), quien = String(it.evaluador || u.nombre).slice(0, 200);
          if (it.tipo === 'amenaza') {
            const n = fA[it.codigo + '|' + it.id];
            if (!n) throw new Error('No existe la amenaza ' + it.id + ' para ' + it.codigo);
            const niv = String(it.nivel || ''), pr = String(it.prepararse || '');
            if (niv && ISH.AMEN_NIVELES.indexOf(niv) < 0) throw new Error('Nivel no válido: ' + niv);
            if (pr && ISH.PREPARARSE.indexOf(pr) < 0) throw new Error('Valor no válido: ' + pr);
            am.getRange(n, 7, 1, 5).setValues([[niv, pr, String(it.obs || '').slice(0, 2000), fecha, quien]]);
            tocadas.push({ uid: it.uid, hoja: am, n: n, tipo: 'amenaza' });
            registrar_(u, 'ISH · Amenaza', red, it.codigo, it.id, niv + (pr ? ' · prepararse: ' + pr : ''));
          } else {
            const n = fR[it.codigo + '|' + it.n];
            if (!n) throw new Error('No existe el ítem ' + it.n + ' para ' + it.codigo);
            const niv = String(it.nivel || '');
            if (niv && ISH.NIVELES.indexOf(niv) < 0) throw new Error('Nivel no válido: ' + niv);
            rl.getRange(n, 9).setValue(niv);
            rl.getRange(n, 13, 1, 3).setValues([[String(it.obs || '').slice(0, 2000), fecha, quien]]);
            tocadas.push({ uid: it.uid, hoja: rl, n: n, tipo: 'item' });
            registrar_(u, 'ISH · Ítem', red, it.codigo, String(it.n), niv || 'sin nivel');
          }
        } catch (err) { resultados.push({ uid: it.uid, ok: false, error: err.message }); }
      });
      if (tocadas.length) {
        SpreadsheetApp.flush();
        tocadas.forEach(t => {
          const v = t.hoja.getRange(t.n, 1, 1, t.tipo === 'amenaza' ? 11 : ISH.NCOL).getDisplayValues()[0];
          resultados.push({ uid: t.uid, ok: true, tipo: t.tipo, fila: t.tipo === 'amenaza' ? ishAmObj_(v) : ishRelObj_(v) });
        });
      }
    });
  } finally { lock.releaseLock(); }
  return { ok: true, resultados: resultados };
}

function ishEstado_(u, req) {
  exigirModulo_(u, 'ISH');
  const r = ishRedes_().filter(x => x.nombre === req.red)[0];
  if (!r) throw new Error('Red no activa');
  if (!cargaEstab_(u, req.red, req.codigo)) throw new Error('Sin permiso de carga en ' + req.codigo);
  if (req.estado && ISH.ESTADOS_REL.indexOf(req.estado) < 0) throw new Error('Estado no válido');
  if (req.modelo && ISH.MODELOS.indexOf(req.modelo) < 0) throw new Error('Modelo no válido');
  return conLock_(() => {
    const sh = SpreadsheetApp.openById(r.id).getSheetByName('ESTABLECIMIENTOS');
    const cod = sh.getRange(2, 1, sh.getLastRow() - 1, 1).getDisplayValues().map(x => x[0]);
    const i = cod.indexOf(req.codigo);
    if (i < 0) throw new Error('Establecimiento inexistente');
    const act = sh.getRange(i + 2, 5, 1, 4).getDisplayValues()[0];
    const nuevo = [req.coordinador != null ? String(req.coordinador).slice(0, 200) || 's/d' : act[0],
                   req.fecha != null ? String(req.fecha) || 's/d' : act[1], req.estado || act[2], req.modelo || act[3]];
    sh.getRange(i + 2, 5, 1, 4).setValues([nuevo]);
    registrar_(u, 'ISH · Datos de la evaluación', req.red, req.codigo, '', nuevo.join(' | '));
    return { ok: true };
  });
}
