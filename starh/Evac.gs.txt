/* ---------- utilidades ---------- */
function evacRedes_() {
  const sh = SpreadsheetApp.getActive().getSheetByName(EVAC.HOJA_REDES);
  if (!sh) return [];
  return sh.getDataRange().getDisplayValues().slice(1)
    .filter(x => x[0] && x[1] && /^activa$/i.test(String(x[2]).trim()))
    .map(x => ({ nombre: x[0].trim(), id: idPlanilla_(x[1]) }));
}
function evacCampos_(ss) {
  return ss.getSheetByName('CAMPOS').getDataRange().getDisplayValues().slice(1)
    .filter(x => x[0] && x[1] && !/^no$/i.test(String(x[7]).trim()))
    .map(x => ({ ficha: x[0], clave: x[1].trim(), seccion: x[2], campo: x[3], tipo: x[4] || 'texto', opciones: String(x[5] || '').split(';').map(s => s.trim()).filter(Boolean), ayuda: x[6] }));
}
// Hojas de datos: fila 1 = nombre visible, fila 2 = clave (no tocar), datos desde la fila 3
function evacCabeceras_(sh, fijas, campos, t) {
  const n = sh.getLastColumn();
  const claves = n ? sh.getRange(2, 1, 1, n).getDisplayValues()[0] : [];
  const fijasClaves = fijas.map((f, i) => '_f' + i);
  if (!n) {
    sh.getRange(1, 1, 2, fijas.length).setValues([fijas, fijasClaves]);
  }
  let col = Math.max(sh.getLastColumn(), fijas.length);
  campos.forEach(c => {
    const i = claves.indexOf(c.clave);
    if (i >= 0) { sh.getRange(1, i + 1).setValue(c.campo); return; }
    col++;
    sh.getRange(1, col, 2, 1).setValues([[c.campo], [c.clave]]);
    claves[col - 1] = c.clave;
  });
  const total = sh.getLastColumn();
  sh.getRange(1, 1, 1, total).setFontWeight('bold').setFontColor('#FFFFFF').setBackground(t.fuerte).setWrap(true).setVerticalAlignment('middle');
  sh.getRange(2, 1, 1, total).setFontColor('#8A979E').setFontSize(8).setBackground('#F3F6F7');
  sh.setFrozenRows(2); sh.setFrozenColumns(Math.min(fijas.length, 4) - (fijas[0] === 'ID' ? 0 : 2));
  sh.setRowHeight(1, 54);
  // listas desplegables
  const cl = sh.getRange(2, 1, 1, total).getDisplayValues()[0];
  campos.forEach(c => {
    const i = cl.indexOf(c.clave); if (i < 0) return;
    const r = sh.getRange(3, i + 1, Math.max(sh.getMaxRows() - 2, 1), 1);
    if (c.tipo === 'lista' && c.opciones.length) r.setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(c.opciones, true).setAllowInvalid(true).build());
    else if (c.tipo === 'sino') r.setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(['Sí', 'No', 's/d'], true).setAllowInvalid(true).build());
    else if (c.tipo === 'telefono') r.setNumberFormat('@');
  });
}

/* ---------- instalación ---------- */
function evacInstalar() {
  const libro = SpreadsheetApp.getActive();
  let rp = libro.getSheetByName(EVAC.HOJA_REDES);
  if (!rp) {
    rp = libro.insertSheet(EVAC.HOJA_REDES);
    rp.getRange(1, 1, 1, 3).setValues([['Red', 'Planilla EVAC (enlace o ID)', 'Estado']]).setFontWeight('bold');
    avisar_('Creé la hoja ' + EVAC.HOJA_REDES + '. Completá una fila por red con el enlace de su planilla y Estado «Activa», y volvé a correr el menú.');
    return;
  }
  const starh = {}; redes_().forEach(r => starh[r.nombre] = r.id);
  const informe = [];
  rp.getDataRange().getDisplayValues().slice(1).filter(x => x[0] && x[1]).forEach(x => {
    const red = x[0].trim(), ss = SpreadsheetApp.openById(idPlanilla_(x[1]));
    if (!starh[red]) throw new Error('No encuentro la planilla STAR-H de ' + red + ' en REDES.');
    const est = SpreadsheetApp.openById(starh[red]).getSheetByName('ESTABLECIMIENTOS').getDataRange().getDisplayValues().slice(1).filter(e => e[0]);
    informe.push(red + ': ' + evacArmar_(ss, red, est));
  });
  avisar_(informe.join('\n') || 'No hay redes en ' + EVAC.HOJA_REDES + '.');
}

function evacArmar_(ss, red, est) {
  const t = /trelew/i.test(red) ? { fuerte: '#0D313F', claro: '#E6EFF2' } : { fuerte: '#B8441A', claro: '#FDEDE4' };

/* ---------- API ---------- */
function evacLeerHoja_(sh) {
  const v = sh.getDataRange().getDisplayValues();
  if (v.length < 2) return { claves: [], filas: [] };
  return { claves: v[1], filas: v.slice(2).filter(x => x[0]) };
}
const EVAC_FIJAS_ = { est: ['codigo', 'nombre', 'tipo', 'red', 'estado', 'actualizado', 'por'], dep: ['id', 'codigo', 'establecimiento', 'nombre', 'estado', 'actualizado', 'por'] };
function evacObj_(claves, x, fijas) {
  const o = {};
  claves.forEach((k, i) => { if (!k) return; const m = k.match(/^_f(\d)$/); o[m ? fijas[Number(m[1])] : k] = x[i]; });
  return o;
}

function evacCatalogo_(u) {
  exigirModulo_(u, 'EVAC');
  const salida = { ok: true, usuario: publico_(u), generado: new Date().toISOString(), estados: EVAC.ESTADOS, defInst: evacDefCampos_(EVAC_INST), defAlb: evacDefCampos_(EVAC_ALB), defCrit: evacDefCampos_(EVAC_CRIT), camas: EVAC_CAMAS, redes: [] };
  evacRedes_().filter(r => veRed_(u, r.nombre)).forEach(r => {
    const ss = SpreadsheetApp.openById(r.id);
    const E = evacLeerHoja_(ss.getSheetByName('ESTABLECIMIENTOS')), D = evacLeerHoja_(ss.getSheetByName('DEPENDENCIAS'));
    const est = E.filas.filter(x => veEstab_(u, x[0])).map(x => Object.assign(evacObj_(E.claves, x, EVAC_FIJAS_.est), { puedeCargar: cargaEstab_(u, r.nombre, x[0]) }));
    if (!est.length) return;
    const cods = est.map(e => e.codigo);
    const inst = evacLeerFijas_(ss, 'INSTITUCIONES');
    salida.redes.push({ nombre: r.nombre, instituciones: inst, albergues: evacLeerFijas_(ss, 'ALBERGUES'), criterios: evacLeerFijas_(ss, 'CRITERIOS'), puedeInst: u.rol === 'Administración' || (u.rol === 'Referente' && veRed_(u, r.nombre)), planilla: u.rol === 'Administración' ? 'https://docs.google.com/spreadsheets/d/' + r.id + '/edit' : '', campos: evacCampos_(ss), establecimientos: est, dependencias: D.filas.filter(x => cods.indexOf(x[1]) >= 0).map(x => evacObj_(D.claves, x, EVAC_FIJAS_.dep)) });
  });
  return salida;
}

function evacGuardar_(u, req) {
  exigirModulo_(u, 'EVAC');
  const r = evacRedes_().filter(x => x.nombre === req.red)[0];
  if (!r) throw new Error('Red no activa');
  if (!cargaEstab_(u, req.red, req.codigo)) throw new Error('Sin permiso de carga en ' + req.codigo);
  const lock = LockService.getScriptLock(); lock.waitLock(30000);
  try {
    const ss = SpreadsheetApp.openById(r.id);
    const esDep = req.ficha === 'Dependencia';
    const sh = ss.getSheetByName(esDep ? 'DEPENDENCIAS' : 'ESTABLECIMIENTOS');
    const L = evacLeerHoja_(sh), claves = L.claves;
    const validas = evacCampos_(ss).filter(c => c.ficha === (esDep ? 'Dependencia' : 'Establecimiento'));
    const tipoDe = {}; validas.forEach(c => tipoDe[c.clave] = c);
    const todas = sh.getRange(1, 1, Math.max(sh.getLastRow(), 2), 1).getDisplayValues().map(x => x[0]);
    let fila, id = '';
    if (esDep) {
      if (req.borrar) {
        const f = todas.indexOf(req.id) + 1; if (f < 3) throw new Error('No existe la dependencia');
        if (sh.getRange(f, 2).getDisplayValue() !== req.codigo) throw new Error('La dependencia no es de ese establecimiento');
        sh.deleteRow(f); registrar_(u, 'EVAC · Quitar dependencia', req.red, req.codigo, req.id, ''); return { ok: true };
      }
      const nombre = String(req.nombre || '').trim(); if (!nombre) throw new Error('Falta el nombre de la dependencia.');
      if (req.id) {
        fila = todas.indexOf(req.id) + 1; if (fila < 3) throw new Error('No existe la dependencia ' + req.id); id = req.id;
        if (sh.getRange(fila, 2).getDisplayValue() !== req.codigo) throw new Error('La dependencia no es de ese establecimiento');
      }
      else {
        let max = 0; todas.forEach(v => { const m = String(v).match(/^D-(\d+)$/); if (m) max = Math.max(max, Number(m[1])); });
        id = 'D-' + String(siguienteNumero_('EVAC|' + r.id + '|D', max)).padStart(3, '0');
        let ult = todas.length; while (ult > 2 && !String(todas[ult - 1]).trim()) ult--;
        fila = ult + 1;
      }
      const est = ss.getSheetByName('ESTABLECIMIENTOS'); const ce = evacLeerHoja_(est).filas.filter(x => x[0] === req.codigo)[0];
      sh.getRange(fila, 1, 1, 4).setValues([[id, req.codigo, ce ? ce[1] : '', nombre.slice(0, 200)]]);
    } else {
      fila = todas.indexOf(req.codigo) + 1; if (fila < 3) throw new Error('Establecimiento inexistente');
    }
    const datos = req.datos || {};
    const fila1 = sh.getRange(fila, 1, 1, claves.length).getValues()[0];
    Object.keys(datos).forEach(k => {
      const i = claves.indexOf(k); if (i < 0 || !tipoDe[k]) return;
      let v = datos[k] == null ? '' : String(datos[k]).trim().slice(0, 2000);
      const c = tipoDe[k];
      if ((c.tipo === 'numero' || c.tipo === 'porcentaje') && v !== '' && v !== 's/d') { const n = Number(v.replace(',', '.')); if (isNaN(n) || n < 0) throw new Error(c.campo + ': tiene que ser un número.'); v = n; }
      if (c.tipo === 'correo' && v && v !== 's/d' && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(v)) throw new Error(c.campo + ': revisá el correo.');
      fila1[i] = v;
    });
    const est = req.estado && EVAC.ESTADOS.indexOf(req.estado) >= 0 ? req.estado : (fila1[4] && fila1[4] !== 'Pendiente' ? fila1[4] : 'En curso');
    fila1[4] = est; fila1[5] = Utilities.formatDate(new Date(), 'America/Argentina/Buenos_Aires', 'dd/MM/yyyy HH:mm'); fila1[6] = u.nombre;
    // las columnas de teléfono quedan como texto
    sh.getRange(fila, 1, 1, claves.length).setValues([fila1]);
    registrar_(u, esDep ? 'EVAC · Ficha de dependencia' : 'EVAC · Ficha del establecimiento', req.red, req.codigo, id, Object.keys(datos).length + ' campos · ' + est);
    return { ok: true, id: id };
  } finally { lock.releaseLock(); }
}

function evacHojaFija_(ss, nombre, def, t, i) {
  let sh = ss.getSheetByName(nombre);
  if (!sh) {
    sh = ss.insertSheet(nombre, i);
    sh.getRange(1, 1, 2, def.length).setValues([def.map(d => d[1]), def.map(d => d[0])]);
  } else {
    // agrega columnas que falten (no borra ni reordena)
    const cl = sh.getRange(2, 1, 1, Math.max(sh.getLastColumn(), 1)).getDisplayValues()[0];
    let col = sh.getLastColumn();
    def.forEach(d => { if (cl.indexOf(d[0]) < 0) { col++; sh.getRange(1, col, 2, 1).setValues([[d[1]], [d[0]]]); } });
  }
  const n = sh.getLastColumn();
  sh.getRange(1, 1, 1, n).setFontWeight('bold').setFontColor('#FFFFFF').setBackground(t.fuerte).setWrap(true).setVerticalAlignment('middle');
  sh.getRange(2, 1, 1, n).setFontColor('#8A979E').setFontSize(8).setBackground('#F3F6F7');
  sh.setFrozenRows(2); sh.setFrozenColumns(2); sh.setRowHeight(1, 54);
  const cl = sh.getRange(2, 1, 1, n).getDisplayValues()[0];
  def.forEach(d => {
    const c = cl.indexOf(d[0]); if (c < 0) return;
    const r = sh.getRange(3, c + 1, Math.max(sh.getMaxRows() - 2, 1), 1);
    if (d[2] === 'lista') r.setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(d[3].split(';'), true).setAllowInvalid(true).build());
    if (d[2] === 'telefono') r.setNumberFormat('@');
  });
  return sh;
}
function evacTraslados_(ss, red) {
  const t = /trelew/i.test(red) ? { fuerte: '#0D313F' } : { fuerte: '#B8441A' };
  evacHojaFija_(ss, 'INSTITUCIONES', EVAC_INST, t, 4).setColumnWidth(2, 260);
  evacHojaFija_(ss, 'ALBERGUES', EVAC_ALB, t, 5).setColumnWidth(2, 240);
  const cr = evacHojaFija_(ss, 'CRITERIOS', EVAC_CRIT, t, 6);
  cr.setColumnWidth(3, 220); cr.setColumnWidth(4, 320); cr.setColumnWidth(5, 220);
  if (cr.getLastRow() < 3) {
    const cl = cr.getRange(2, 1, 1, cr.getLastColumn()).getDisplayValues()[0], k = ['sentido', 'grupo', 'criterio', 'destino', 'condiciones', 'decide', 'informa', 'fuente'];
    const filas = EVAC_CRIT_BASE.map((b, i) => { const v = cl.map(() => ''); v[cl.indexOf('id')] = 'C-' + String(i + 1).padStart(2, '0'); k.forEach((kk, j) => v[cl.indexOf(kk)] = b[j]); v[cl.indexOf('estado')] = 'Propuesto'; return v; });
    cr.getRange(3, 1, filas.length, cl.length).setValues(filas).setWrap(true).setVerticalAlignment('top');
    siguienteNumero_('EVAC|' + ss.getId() + '|C', filas.length);
  }
}

function evacDefCampos_(def) { return def.map(d => ({ clave: d[0], campo: d[1], tipo: d[2], opciones: d[3] ? d[3].split(';') : [] })); }
function evacSiguiente_(sh, pref, ancho) {
  let max = 0;
  sh.getRange(1, 1, Math.max(sh.getLastRow(), 1), 1).getDisplayValues().forEach(x => { const m = String(x[0]).match(new RegExp('^' + pref + '-(\\d+)$')); if (m) max = Math.max(max, Number(m[1])); });
  return pref + '-' + String(siguienteNumero_('EVAC|' + sh.getParent().getId() + '|' + pref, max)).padStart(ancho, '0');
}
function evacUltimaFila_(sh) {
  const a = sh.getRange(1, 1, Math.max(sh.getLastRow(), 2), 1).getDisplayValues();
  for (let i = a.length - 1; i >= 2; i--) if (String(a[i][0]).trim()) return i + 1;
  return 2;
}
// Guarda una fila (alta, cambio o baja) en INSTITUCIONES
function evacGuardarFila_(u, req, hoja, def, pref, ancho, puede) {
  const r = evacRedes_().filter(x => x.nombre === req.red)[0];
  if (!r) throw new Error('Red no activa');
  if (!puede(r)) throw new Error('Sin permiso de carga.');
  const lock = LockService.getScriptLock(); lock.waitLock(30000);
  try {
    const ss = SpreadsheetApp.openById(r.id);
    let sh = ss.getSheetByName(hoja); if (!sh) { evacTraslados_(ss, r.nombre); sh = ss.getSheetByName(hoja); }
    const claves = sh.getRange(2, 1, 1, sh.getLastColumn()).getDisplayValues()[0];
    const ids = sh.getRange(1, 1, Math.max(sh.getLastRow(), 2), 1).getDisplayValues().map(x => x[0]);
    if (req.borrar) {
      const f = ids.indexOf(req.id) + 1; if (f < 3) throw new Error('No existe ' + req.id);
      sh.deleteRow(f); registrar_(u, 'EVAC · Quitar ' + hoja.toLowerCase(), req.red, req.datos && req.datos.codigo || '', req.id, ''); return { ok: true };
    }
    let fila, id = req.id;
    if (id) { fila = ids.indexOf(id) + 1; if (fila < 3) throw new Error('No existe ' + id); }
    else { id = evacSiguiente_(sh, pref, ancho); fila = evacUltimaFila_(sh) + 1; }
    const tipo = {}; def.forEach(d => tipo[d[0]] = d);
    const v = fila <= sh.getLastRow() ? sh.getRange(fila, 1, 1, claves.length).getValues()[0] : claves.map(() => '');
    const datos = req.datos || {};
    Object.keys(datos).forEach(k => {
      const i = claves.indexOf(k), d = tipo[k]; if (i < 0 || !d || k === 'id') return;
      let x = datos[k] == null ? '' : String(datos[k]).trim().slice(0, 2000);
      if (d[2] === 'numero' && x !== '' && x !== 's/d') { const n = Number(x.replace(',', '.')); if (isNaN(n) || n < 0) throw new Error(d[1] + ': tiene que ser un número.'); x = n; }
      if (d[2] === 'correo' && x && x !== 's/d' && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(x)) throw new Error(d[1] + ': revisá el correo.');
      v[i] = x;
    });
    v[claves.indexOf('id')] = id;
    const ia = claves.indexOf('actualizado'), ip = claves.indexOf('por');
    if (ia >= 0) v[ia] = Utilities.formatDate(new Date(), 'America/Argentina/Buenos_Aires', 'dd/MM/yyyy HH:mm');
    if (ip >= 0) v[ip] = u.nombre;
    sh.getRange(fila, 1, 1, claves.length).setValues([v]);
    registrar_(u, 'EVAC · ' + ({ INSTITUCIONES: 'Institución receptora', ALBERGUES: 'Albergue', CRITERIOS: 'Criterio de derivación' }[hoja] || hoja), req.red, datos.codigo || '', id, String(datos.nombre || '').slice(0, 80));
    return { ok: true, id: id };
  } finally { lock.releaseLock(); }
}
function evacInstitucion_(u, req) {
  exigirModulo_(u, 'EVAC');
  return evacGuardarFila_(u, req, 'INSTITUCIONES', EVAC_INST, 'I', 3, r => u.rol === 'Administración' || (u.rol === 'Referente' && veRed_(u, r.nombre)));
}
function evacAlbergue_(u, req) {
  exigirModulo_(u, 'EVAC');
  return evacGuardarFila_(u, req, 'ALBERGUES', EVAC_ALB, 'A', 3, r => u.rol === 'Administración' || (u.rol === 'Referente' && veRed_(u, r.nombre)));
}
function evacCriterio_(u, req) {
  exigirModulo_(u, 'EVAC');
  return evacGuardarFila_(u, req, 'CRITERIOS', EVAC_CRIT, 'C', 2, r => u.rol === 'Administración' || (u.rol === 'Referente' && veRed_(u, r.nombre)));
}
function evacLeerFijas_(ss, hoja) {
  const sh = ss.getSheetByName(hoja); if (!sh) return [];
  const L = evacLeerHoja_(sh);
  return L.filas.map(x => { const o = {}; L.claves.forEach((k, i) => { if (k) o[k] = x[i]; }); return o; });
}
