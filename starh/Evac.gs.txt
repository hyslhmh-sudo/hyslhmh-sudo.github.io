/* ---------- utilidades ---------- */
function evacRedes_() {
  const sh = SpreadsheetApp.getActive().getSheetByName(EVAC.HOJA_REDES);
  if (!sh) return [];
  return sh.getDataRange().getDisplayValues().slice(1)
    .filter(x => x[0] && x[1] && /^activa$/i.test(String(x[2]).trim()))
    .map(x => ({ nombre: x[0].trim(), id: idPlanilla_(x[1]) }));
}
function evacCampos_(ss) {
  return ss.getSheetByName('CAMPOS').getDataRange().getDisplayValues().slice(1)
    .filter(x => x[0] && x[1] && !/^no$/i.test(String(x[7]).trim()))
    .map(x => ({ ficha: x[0], clave: x[1].trim(), seccion: x[2], campo: x[3], tipo: x[4] || 'texto', opciones: String(x[5] || '').split(';').map(s => s.trim()).filter(Boolean), ayuda: x[6] }));
}
// Hojas de datos: fila 1 = nombre visible, fila 2 = clave (no tocar), datos desde la fila 3
function evacCabeceras_(sh, fijas, campos, t) {
  const n = sh.getLastColumn();
  const claves = n ? sh.getRange(2, 1, 1, n).getDisplayValues()[0] : [];
  const fijasClaves = fijas.map((f, i) => '_f' + i);
  if (!n) {
    sh.getRange(1, 1, 2, fijas.length).setValues([fijas, fijasClaves]);
  }
  let col = Math.max(sh.getLastColumn(), fijas.length);
  campos.forEach(c => {
    const i = claves.indexOf(c.clave);
    if (i >= 0) { sh.getRange(1, i + 1).setValue(c.campo); return; }
    col++;
    sh.getRange(1, col, 2, 1).setValues([[c.campo], [c.clave]]);
    claves[col - 1] = c.clave;
  });
  const total = sh.getLastColumn();
  sh.getRange(1, 1, 1, total).setFontWeight('bold').setFontColor('#FFFFFF').setBackground(t.fuerte).setWrap(true).setVerticalAlignment('middle');
  sh.getRange(2, 1, 1, total).setFontColor('#8A979E').setFontSize(8).setBackground('#F3F6F7');
  sh.setFrozenRows(2); sh.setFrozenColumns(Math.min(fijas.length, 4) - (fijas[0] === 'ID' ? 0 : 2));
  sh.setRowHeight(1, 54);
  // listas desplegables
  const cl = sh.getRange(2, 1, 1, total).getDisplayValues()[0];
  campos.forEach(c => {
    const i = cl.indexOf(c.clave); if (i < 0) return;
    const r = sh.getRange(3, i + 1, Math.max(sh.getMaxRows() - 2, 1), 1);
    if (c.tipo === 'lista' && c.opciones.length) r.setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(c.opciones, true).setAllowInvalid(true).build());
    else if (c.tipo === 'sino') r.setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(['Sí', 'No', 's/d'], true).setAllowInvalid(true).build());
    else if (c.tipo === 'telefono') r.setNumberFormat('@');
  });
}

/* ---------- instalación ---------- */
function evacInstalar() {
  const libro = SpreadsheetApp.getActive();
  let rp = libro.getSheetByName(EVAC.HOJA_REDES);
  if (!rp) {
    rp = libro.insertSheet(EVAC.HOJA_REDES);
    rp.getRange(1, 1, 1, 3).setValues([['Red', 'Planilla EVAC (enlace o ID)', 'Estado']]).setFontWeight('bold');
    avisar_('Creé la hoja ' + EVAC.HOJA_REDES + '. Completá una fila por red con el enlace de su planilla y Estado «Activa», y volvé a correr el menú.');
    return;
  }
  const starh = {}; redes_().forEach(r => starh[r.nombre] = r.id);
  const informe = [];
  rp.getDataRange().getDisplayValues().slice(1).filter(x => x[0] && x[1]).forEach(x => {
    const red = x[0].trim(), ss = SpreadsheetApp.openById(idPlanilla_(x[1]));
    if (!starh[red]) throw new Error('No encuentro la planilla STAR-H de ' + red + ' en REDES.');
    const est = SpreadsheetApp.openById(starh[red]).getSheetByName('ESTABLECIMIENTOS').getDataRange().getDisplayValues().slice(1).filter(e => e[0]);
    informe.push(red + ': ' + evacArmar_(ss, red, est));
  });
  avisar_(informe.join('\n') || 'No hay redes en ' + EVAC.HOJA_REDES + '.');
}

  ay.getRange('A1').setFontSize(15).setFontWeight('bold').setFontColor(t.fuerte);
  ['A4', 'A8', 'A12', 'A15'].forEach(a => ay.getRange(a).setFontWeight('bold').setFontColor(t.fuerte));

  // CAMPOS (solo la primera vez: después es de ella)
  let ca = ss.getSheetByName('CAMPOS');
  if (!ca) {
    ca = hoja('CAMPOS', 1);
    ca.getRange(1, 1, 1, 8).setValues([['Ficha', 'Clave', 'Sección', 'Campo', 'Tipo', 'Opciones (separadas por ;)', 'Ayuda', 'Activo']]);
    ca.getRange(2, 1, EVAC.CAMPOS.length, 8).setValues(EVAC.CAMPOS.map(c => c.concat(['Sí'])));
    ca.getRange(1, 1, 1, 8).setFontWeight('bold').setFontColor('#FFFFFF').setBackground(t.fuerte);
    ca.setFrozenRows(1); ca.setColumnWidth(2, 130); ca.setColumnWidth(3, 220); ca.setColumnWidth(4, 340); ca.setColumnWidth(6, 320); ca.setColumnWidth(7, 320);
    ca.getRange('A2:A').setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(['Establecimiento', 'Dependencia'], true).build());
    ca.getRange('E2:E').setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(['texto', 'largo', 'numero', 'porcentaje', 'fecha', 'sino', 'lista', 'multi', 'telefono', 'correo'], true).build());
    ca.getRange('H2:H').setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(['Sí', 'No'], true).build());
  }
  const campos = evacCampos_(ss);

  // ESTABLECIMIENTOS
  const es = hoja('ESTABLECIMIENTOS', 2);
  evacCabeceras_(es, EVAC.FIJAS_EST, campos.filter(c => c.ficha === 'Establecimiento'), t);
  const ya = es.getLastRow() > 2 ? es.getRange(3, 1, es.getLastRow() - 2, 1).getDisplayValues().map(x => x[0]) : [];
  const nuevos = est.filter(e => ya.indexOf(e[0]) < 0).map(e => [e[0], e[3], e[2], red, 'Pendiente', '', '']);
  if (nuevos.length) es.getRange(Math.max(es.getLastRow(), 2) + 1, 1, nuevos.length, 7).setValues(nuevos);
  es.getRange(3, 5, Math.max(es.getMaxRows() - 2, 1), 1).setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(EVAC.ESTADOS, true).build());
  es.setColumnWidth(2, 260);

  // DEPENDENCIAS
  const de = hoja('DEPENDENCIAS', 3);
  evacCabeceras_(de, EVAC.FIJAS_DEP, campos.filter(c => c.ficha === 'Dependencia'), t);
  de.getRange(3, 5, Math.max(de.getMaxRows() - 2, 1), 1).setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(EVAC.ESTADOS, true).build());
  de.setColumnWidth(3, 220); de.setColumnWidth(4, 220);

  ['Hoja 1', 'Hoja1', 'Sheet1'].forEach(nm => { const s = ss.getSheetByName(nm); if (s && ss.getSheets().length > 1) ss.deleteSheet(s); });
  return est.length + ' establecimientos (' + nuevos.length + ' nuevos) · ' + campos.length + ' campos activos.';
}

/* ---------- API ---------- */
function evacLeerHoja_(sh) {
  const v = sh.getDataRange().getDisplayValues();
  if (v.length < 2) return { claves: [], filas: [] };
  return { claves: v[1], filas: v.slice(2).filter(x => x[0]) };
}
const EVAC_FIJAS_ = { est: ['codigo', 'nombre', 'tipo', 'red', 'estado', 'actualizado', 'por'], dep: ['id', 'codigo', 'establecimiento', 'nombre', 'estado', 'actualizado', 'por'] };
function evacObj_(claves, x, fijas) {
  const o = {};
  claves.forEach((k, i) => { if (!k) return; const m = k.match(/^_f(\d)$/); o[m ? fijas[Number(m[1])] : k] = x[i]; });
  return o;
}

function evacCatalogo_(u) {
  exigirModulo_(u, 'EVAC');
  const salida = { ok: true, usuario: publico_(u), generado: new Date().toISOString(), estados: EVAC.ESTADOS, redes: [] };
  evacRedes_().filter(r => veRed_(u, r.nombre)).forEach(r => {
    const ss = SpreadsheetApp.openById(r.id);
    const E = evacLeerHoja_(ss.getSheetByName('ESTABLECIMIENTOS')), D = evacLeerHoja_(ss.getSheetByName('DEPENDENCIAS'));
    const est = E.filas.filter(x => veEstab_(u, x[0])).map(x => Object.assign(evacObj_(E.claves, x, EVAC_FIJAS_.est), { puedeCargar: cargaEstab_(u, r.nombre, x[0]) }));
    if (!est.length) return;
    const cods = est.map(e => e.codigo);
    salida.redes.push({ nombre: r.nombre, campos: evacCampos_(ss), establecimientos: est, dependencias: D.filas.filter(x => cods.indexOf(x[1]) >= 0).map(x => evacObj_(D.claves, x, EVAC_FIJAS_.dep)) });
  });
  return salida;
}

function evacGuardar_(u, req) {
  exigirModulo_(u, 'EVAC');
  const r = evacRedes_().filter(x => x.nombre === req.red)[0];
  if (!r) throw new Error('Red no activa');
  if (!cargaEstab_(u, req.red, req.codigo)) throw new Error('Sin permiso de carga en ' + req.codigo);
  const lock = LockService.getScriptLock(); lock.waitLock(30000);
  try {
    const ss = SpreadsheetApp.openById(r.id);
    const esDep = req.ficha === 'Dependencia';
    const sh = ss.getSheetByName(esDep ? 'DEPENDENCIAS' : 'ESTABLECIMIENTOS');
    const L = evacLeerHoja_(sh), claves = L.claves;
    const validas = evacCampos_(ss).filter(c => c.ficha === (esDep ? 'Dependencia' : 'Establecimiento'));
    const tipoDe = {}; validas.forEach(c => tipoDe[c.clave] = c);
    const todas = sh.getRange(1, 1, Math.max(sh.getLastRow(), 2), 1).getDisplayValues().map(x => x[0]);
    let fila, id = '';
    if (esDep) {
      if (req.borrar) {
        const f = todas.indexOf(req.id) + 1; if (f < 3) throw new Error('No existe la dependencia');
        if (sh.getRange(f, 2).getDisplayValue() !== req.codigo) throw new Error('La dependencia no es de ese establecimiento');
        sh.deleteRow(f); registrar_(u, 'EVAC · Quitar dependencia', req.red, req.codigo, req.id, ''); return { ok: true };
      }
      const nombre = String(req.nombre || '').trim(); if (!nombre) throw new Error('Falta el nombre de la dependencia.');
      if (req.id) { fila = todas.indexOf(req.id) + 1; if (fila < 3) throw new Error('No existe la dependencia ' + req.id); id = req.id; }
      else {
        let max = 0; todas.forEach(v => { const m = String(v).match(/^D-(\d+)$/); if (m) max = Math.max(max, Number(m[1])); });
        id = 'D-' + ('00' + (max + 1)).slice(-3);
        let ult = todas.length; while (ult > 2 && !String(todas[ult - 1]).trim()) ult--;
        fila = ult + 1;
      }
      const est = ss.getSheetByName('ESTABLECIMIENTOS'); const ce = evacLeerHoja_(est).filas.filter(x => x[0] === req.codigo)[0];
      sh.getRange(fila, 1, 1, 4).setValues([[id, req.codigo, ce ? ce[1] : '', nombre.slice(0, 200)]]);
    } else {
      fila = todas.indexOf(req.codigo) + 1; if (fila < 3) throw new Error('Establecimiento inexistente');
    }
    const datos = req.datos || {};
    const fila1 = sh.getRange(fila, 1, 1, claves.length).getValues()[0];
    Object.keys(datos).forEach(k => {
      const i = claves.indexOf(k); if (i < 0 || !tipoDe[k]) return;
      let v = datos[k] == null ? '' : String(datos[k]).trim().slice(0, 2000);
      const c = tipoDe[k];
      if ((c.tipo === 'numero' || c.tipo === 'porcentaje') && v !== '' && v !== 's/d') { const n = Number(v.replace(',', '.')); if (isNaN(n) || n < 0) throw new Error(c.campo + ': tiene que ser un número.'); v = n; }
      if (c.tipo === 'correo' && v && v !== 's/d' && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(v)) throw new Error(c.campo + ': revisá el correo.');
      fila1[i] = v;
    });
    const est = req.estado && EVAC.ESTADOS.indexOf(req.estado) >= 0 ? req.estado : (fila1[4] && fila1[4] !== 'Pendiente' ? fila1[4] : 'En curso');
    fila1[4] = est; fila1[5] = Utilities.formatDate(new Date(), 'America/Argentina/Buenos_Aires', 'dd/MM/yyyy HH:mm'); fila1[6] = u.nombre;
    // las columnas de teléfono quedan como texto
    sh.getRange(fila, 1, 1, claves.length).setValues([fila1]);
    registrar_(u, esDep ? 'EVAC · Ficha de dependencia' : 'EVAC · Ficha del establecimiento', req.red, req.codigo, id, Object.keys(datos).length + ' campos · ' + est);
    return { ok: true, id: id };
  } finally { lock.releaseLock(); }
}
