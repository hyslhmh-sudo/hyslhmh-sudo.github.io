function phrAplica_(aplicaA, tipo) {
  if (aplicaA === 'Todos') return true;
  if (aplicaA === 'Hospitales') return tipo === 'Hospital';
  if (aplicaA === 'Hospitales y Sede ex ADOS') return tipo === 'Hospital' || tipo === 'Sede ex ADOS';
  return true;
}

/* =========================================================
 *  INSTALACIÓN: arma la planilla PHR de cada red listada en REDES PHR
 *  (si la hoja RELEVAMIENTO ya tiene datos, no la toca)
 * ========================================================= */
function phrInstalar() {
  const libro = SpreadsheetApp.getActive();
  const rp = libro.getSheetByName(PHR.HOJA_REDES);
  if (!rp) throw new Error('Falta la hoja ' + PHR.HOJA_REDES);
  const starh = {};
  redes_().forEach(r => starh[r.nombre] = r.id);
  const filas = rp.getDataRange().getDisplayValues().slice(1).filter(x => x[0] && x[1]);
  const informe = [];
  filas.forEach(x => {
    const red = x[0].trim(), id = idPlanilla_(x[1]);
    const ss = SpreadsheetApp.openById(id);
    const rel = ss.getSheetByName('RELEVAMIENTO');
    if (rel && rel.getLastRow() > 1) { informe.push(red + ': ya estaba armada, no se tocó.'); return; }
    if (!starh[red]) throw new Error('No encuentro la planilla STAR-H de ' + red + ' en REDES (de ahí salen los establecimientos).');
    const est = SpreadsheetApp.openById(starh[red]).getSheetByName('ESTABLECIMIENTOS').getDataRange().getDisplayValues().slice(1).filter(e => e[0]);
    phrArmar_(ss, red, est);
    informe.push(red + ': ' + est.length + ' establecimientos · ' + (est.length * PHR_ITEMS.length) + ' ítems.');
  });
  avisar_(informe.join('\n') || 'No hay redes en ' + PHR.HOJA_REDES + '.');
}

  rl.getRange(2, 15, n, 1).setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(PHR.PRIORIDADES, true).build());

  rl.getRange(2, 1, n, 17).setVerticalAlignment('top');
  rl.getRange(2, 7, n, 1).setWrap(true);
  const reglas = [
    ['Cumple', '#C8E6C9', '#1E5223'], ['Parcial', '#FFF2B3', '#5E4B00'], ['No cumple', '#F4B4B4', '#7A1F1F']
  ].map(c => SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo(c[0]).setBackground(c[1]).setFontColor(c[2]).setRanges([rl.getRange(2, 10, n, 1)]).build());
  reglas.push(SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('No aplica').setFontColor('#8A979E').setRanges([rl.getRange(2, 9, n, 1)]).build());
  rl.setConditionalFormatRules(reglas);
  rl.getRange('K1:K').protect().setDescription('Cálculo automático').setWarningOnly(true);

  // TABLERO (por establecimiento y sección)
  const tb = hoja('TABLERO', 4);
  tb.clear();
  tb.getRange('A1').setValue('PHR · ' + red + ' · Cumplimiento por establecimiento y sección').setFontSize(14).setFontWeight('bold').setFontColor(t.fuerte);
  tb.getRange('A2').setValue('Promedio de los ítems que aplican y tienen estado (Cumple = 100 %, Parcial = 50 %, No cumple = 0 %). «s/d»: sin datos todavía.').setFontColor('#6B7A82');
  const encab = ['Código', 'Establecimiento', 'Tipo', 'Estado', 'Ítems relevados', 'Total'].concat(PHR.SECCIONES.map(s => s[0] + ' · ' + s[1]));
  tb.getRange(4, 1, 1, encab.length).setValues([encab]);
  const f = [];
  est.forEach((e, i) => {
    const r = 5 + i;
    const fila = ['=ESTABLECIMIENTOS!A' + (i + 2), '=ESTABLECIMIENTOS!D' + (i + 2), '=ESTABLECIMIENTOS!C' + (i + 2), '=ESTABLECIMIENTOS!G' + (i + 2),
      '=COUNTIFS(RELEVAMIENTO!$A:$A,$A' + r + ',RELEVAMIENTO!$J:$J,"<>")&" de "&COUNTIFS(RELEVAMIENTO!$A:$A,$A' + r + ',RELEVAMIENTO!$I:$I,"Sí")',
      '=IFERROR(AVERAGEIFS(RELEVAMIENTO!$K:$K,RELEVAMIENTO!$A:$A,$A' + r + '),"s/d")'];
    PHR.SECCIONES.forEach(s => fila.push('=IFERROR(AVERAGEIFS(RELEVAMIENTO!$K:$K,RELEVAMIENTO!$A:$A,$A' + r + ',RELEVAMIENTO!$F:$F,"' + s[0] + ' · *"),"s/d")'));
    f.push(fila);
  });
  tb.getRange(5, 1, f.length, encab.length).setFormulas(f);
  const tot = ['', 'Total de la red', '', '', '', '=IFERROR(AVERAGE(RELEVAMIENTO!K2:K),"s/d")'];
  PHR.SECCIONES.forEach(s => tot.push('=IFERROR(AVERAGEIFS(RELEVAMIENTO!$K:$K,RELEVAMIENTO!$F:$F,"' + s[0] + ' · *"),"s/d")'));
  tb.getRange(5 + f.length, 1, 1, encab.length).setFormulas([tot]).setFontWeight('bold');
  tb.getRange(4, 1, 1, encab.length).setFontWeight('bold').setFontColor('#FFFFFF').setBackground(t.fuerte).setWrap(true).setVerticalAlignment('middle');
  tb.setFrozenRows(4); tb.setFrozenColumns(2);
  tb.setColumnWidth(2, 300); tb.setColumnWidths(7, PHR.SECCIONES.length, 110);
  const rango = tb.getRange(5, 6, f.length + 1, PHR.SECCIONES.length + 1);
  rango.setNumberFormat('0%').setHorizontalAlignment('center');
  tb.setConditionalFormatRules([SpreadsheetApp.newConditionalFormatRule().setGradientMinpointWithValue('#F4B4B4', SpreadsheetApp.InterpolationType.NUMBER, '0')
    .setGradientMidpointWithValue('#FFF2B3', SpreadsheetApp.InterpolationType.NUMBER, '0.5').setGradientMaxpointWithValue('#C8E6C9', SpreadsheetApp.InterpolationType.NUMBER, '1')
    .setRanges([rango]).build()]);
  tb.protect().setDescription('Tablero con fórmulas').setWarningOnly(true);

  // limpiar la hoja vacía inicial
  ['Hoja 1', 'Hoja1', 'Sheet1'].forEach(nm => { const s = ss.getSheetByName(nm); if (s && ss.getSheets().length > 1) ss.deleteSheet(s); });
}

/* =========================================================
 *  API PHR (la llaman phr.html y phr-tablero.html)
 * ========================================================= */
function phrRedes_() {
  const sh = SpreadsheetApp.getActive().getSheetByName(PHR.HOJA_REDES);
  if (!sh) return [];
  return sh.getDataRange().getDisplayValues().slice(1)
    .filter(x => x[0] && x[1] && /^activa$/i.test(String(x[2]).trim()))
    .map(x => ({ nombre: x[0].trim(), id: idPlanilla_(x[1]) }));
}

function phrRelObj_(x) {
  return { codigo: x[0], estab: x[1], red: x[2], tipo: x[3], id: x[4], seccion: x[5], item: x[6], ref: x[7], aplica: x[8], estado: x[9], puntaje: x[10],
           evidencia: x[11], brecha: x[12], responsable: x[13], prioridad: x[14], fecha: x[15], relevador: x[16] };
}

function phrCatalogo_(u) {
  exigirModulo_(u, 'PHR');
  const salida = { ok: true, usuario: publico_(u), generado: new Date().toISOString(), secciones: PHR.SECCIONES, redes: [] };
  phrRedes_().filter(r => veRed_(u, r.nombre)).forEach(r => {
    const ss = SpreadsheetApp.openById(r.id);
    const est = ss.getSheetByName('ESTABLECIMIENTOS').getDataRange().getDisplayValues().slice(1)
      .filter(x => x[0] && veEstab_(u, x[0]))
      .map(x => ({ codigo: x[0], tipo: x[2], nombre: x[3], referente: x[4], fecha: x[5], estado: x[6], puedeCargar: cargaEstab_(u, r.nombre, x[0]) }));
    if (!est.length) return;
    const cods = est.map(x => x.codigo);
    const items = ss.getSheetByName('ITEMS').getDataRange().getDisplayValues().slice(1).filter(x => x[0])
      .map(x => ({ id: x[0], seccion: x[1], item: x[2], ref: x[3], aplicaA: x[4], evidencia: x[5] }));
    const rel = ss.getSheetByName('RELEVAMIENTO').getDataRange().getDisplayValues().slice(1)
      .filter(x => x[0] && cods.indexOf(x[0]) >= 0).map(phrRelObj_);
    salida.redes.push({ nombre: r.nombre, establecimientos: est, items: items, relevamiento: rel });
  });
  return salida;
}

function phrGuardar_(u, items) {
  exigirModulo_(u, 'PHR');
  if (!items.length) return { ok: true, resultados: [] };
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  const resultados = [];
  try {
    const mapa = {}; phrRedes_().forEach(r => mapa[r.nombre] = r);
    const porRed = {}; items.forEach(it => (porRed[it.red] = porRed[it.red] || []).push(it));
    Object.keys(porRed).forEach(red => {
      const r = mapa[red];
      if (!r) { porRed[red].forEach(it => resultados.push({ uid: it.uid, ok: false, error: 'Red no activa: ' + red })); return; }
      const sh = SpreadsheetApp.openById(r.id).getSheetByName('RELEVAMIENTO');
      const claves = sh.getRange(2, 1, Math.max(sh.getLastRow() - 1, 1), 5).getDisplayValues();
      const fila = {}; claves.forEach((k, i) => { if (k[0]) fila[k[0] + '|' + k[4]] = i + 2; });
      const tocadas = [];
      porRed[red].forEach(it => {
        try {
          if (!cargaEstab_(u, red, it.codigo)) throw new Error('Sin permiso de carga en ' + it.codigo);
          const n = fila[it.codigo + '|' + it.id];
          if (!n) throw new Error('No existe el ítem ' + it.id + ' para ' + it.codigo);
          const aplica = it.aplica === 'No aplica' ? 'No aplica' : 'Sí';
          const estado = aplica === 'No aplica' ? '' : String(it.estado || '');
          if (estado && PHR.ESTADOS.indexOf(estado) < 0) throw new Error('Estado no válido: ' + estado);
          const prioridad = String(it.prioridad || '');
          if (prioridad && PHR.PRIORIDADES.indexOf(prioridad) < 0) throw new Error('Prioridad no válida: ' + prioridad);
          const fecha = it.fecha || Utilities.formatDate(new Date(), 'America/Argentina/Buenos_Aires', 'dd/MM/yyyy');
          sh.getRange(n, PHR.C.aplica, 1, 2).setValues([[aplica, estado]]);
          sh.getRange(n, PHR.C.evidencia, 1, 6).setValues([[String(it.evidencia || '').slice(0, 2000), String(it.brecha || '').slice(0, 2000),
            String(it.responsable || '').slice(0, 300), prioridad, fecha, String(it.relevador || u.nombre).slice(0, 200)]]);
          tocadas.push({ uid: it.uid, n: n });
          registrar_(u, 'PHR · Relevamiento', red, it.codigo, it.id, aplica === 'No aplica' ? 'No aplica' : (estado || 'sin estado') + (prioridad ? ' · prioridad ' + prioridad : ''));
        } catch (err) { resultados.push({ uid: it.uid, ok: false, error: err.message }); }
      });
      if (tocadas.length) {
        SpreadsheetApp.flush();
        tocadas.forEach(t => resultados.push({ uid: t.uid, ok: true, fila: phrRelObj_(sh.getRange(t.n, 1, 1, 17).getDisplayValues()[0]) }));
      }
    });
  } finally { lock.releaseLock(); }
  return { ok: true, resultados: resultados };
}

function phrEstado_(u, req) {
  exigirModulo_(u, 'PHR');
  const r = phrRedes_().filter(x => x.nombre === req.red)[0];
  if (!r) throw new Error('Red no activa');
  if (!cargaEstab_(u, req.red, req.codigo)) throw new Error('Sin permiso de carga en ' + req.codigo);
  if (req.estado && PHR.ESTADOS_REL.indexOf(req.estado) < 0) throw new Error('Estado no válido');
  const sh = SpreadsheetApp.openById(r.id).getSheetByName('ESTABLECIMIENTOS');
  const cod = sh.getRange(2, 1, sh.getLastRow() - 1, 1).getDisplayValues().map(x => x[0]);
  const i = cod.indexOf(req.codigo);
  if (i < 0) throw new Error('Establecimiento inexistente');
  const act = sh.getRange(i + 2, 5, 1, 3).getDisplayValues()[0];
  const nuevo = [req.referente != null ? String(req.referente).slice(0, 200) || 's/d' : act[0],
                 req.fecha != null ? String(req.fecha) || 's/d' : act[1], req.estado || act[2]];
  sh.getRange(i + 2, 5, 1, 3).setValues([nuevo]);
  registrar_(u, 'PHR · Datos del relevamiento', req.red, req.codigo, '', nuevo.join(' | '));
  return { ok: true };
}

function phrTablero_(u) {
  exigirModulo_(u, 'PHR');
  const salida = { ok: true, usuario: publico_(u), generado: new Date().toISOString(), secciones: PHR.SECCIONES, redes: [] };
  phrRedes_().filter(r => veRed_(u, r.nombre)).forEach(r => {
    const ss = SpreadsheetApp.openById(r.id);
    const est = ss.getSheetByName('ESTABLECIMIENTOS').getDataRange().getDisplayValues().slice(1)
      .filter(x => x[0] && veEstab_(u, x[0])).map(x => ({ codigo: x[0], tipo: x[2], nombre: x[3], referente: x[4], fecha: x[5], estado: x[6] }));
    if (!est.length) return;
    const cods = est.map(x => x.codigo);
    // fila compacta: código, id, sección (letra), ítem, aplica, estado, prioridad
    const filas = ss.getSheetByName('RELEVAMIENTO').getDataRange().getDisplayValues().slice(1)
      .filter(x => x[0] && cods.indexOf(x[0]) >= 0).map(x => [x[0], x[4], String(x[5]).charAt(0), x[6], x[8], x[9], x[14]]);
    salida.redes.push({ nombre: r.nombre, establecimientos: est, filas: filas });
  });
  return salida;
}
